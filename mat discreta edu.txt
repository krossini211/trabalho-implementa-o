void arranjo() {
    int n, k, i, fat_n = 1, fat_nk = 1;
    
    printf("Digite n: ");
    scanf("%d", &n);
    printf("Digite k: ");
    scanf("%d", &k);
    
    for(i = 1; i <= n; i++) fat_n *= i;
    for(i = 1; i <= (n - k); i++) fat_nk *= i;
    
    printf("Resultado do Arranjo: %d\n", fat_n / fat_nk);
}

void permutacao() {
    int n, i, fat = 1;
    
    printf("Digite n: ");
    scanf("%d", &n);
    
    for(i = 1; i <= n; i++) {
        fat *= i;
    }
    printf("Resultado: %d\n", fat);
}

void combinacao() {
    int n, k, i, fat_n = 1, fat_k = 1, fat_nk = 1;
    
    printf("Digite n: ");
    scanf("%d", &n);
    printf("Digite k: ");
    scanf("%d", &k);
    
    for(i = 1; i <= n; i++) fat_n *= i;
    for(i = 1; i <= k; i++) fat_k *= i;
    for(i = 1; i <= (n - k); i++) fat_nk *= i;
    
    printf("Resultado: %d\n", fat_n / (fat_k * fat_nk));
}
